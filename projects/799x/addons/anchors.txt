# Anchor hooks for the main (9v3) map (FF_JASS_ANCHORS; verify-map --anchors gates them).
# The build changes the map's own functions at fixed lines: each `insert` line goes right after the `anchor` line
# inside the named `function`, and a `replace` line takes the anchor line's place. The anchor must match a whole
# line (whitespace-trimmed) EXACTLY ONCE in that function, or the build fails: a new base map that changes one of
# these triggers has to be looked at, not shipped with silent zeros.
#   function <name>   starts the hooks for that function (anchors are scoped by function: BASE destroy and
#                     player destroyed share identical lines)
#   after <line>      optional, before an anchor: a line found exactly once in the function; the anchor is then the
#                     first match after it (for a line the function repeats, e.g. in both branches of an if)
#   anchor <line>     the base script's line to hook
#   replace <line>    optional: the line that takes the anchor's place
#   insert <line>     lines to add after the anchor (indented like the anchor); a hook needs a replace or an insert
# The common addons (league/) come before the map's functions in the script, so inserts can use their globals and
# call their functions, and the map's own functions; never a map addon's (projects/<map>/addons/) function.
#
# Stats (league/15-stats.j): gold the map moves itself, which the score screen never sees.
# Each insert repeats the hooked call's own amount and credits that call's real sender and receiver slot.

# Pooling (FFP_Transfer moves every -pool / autopool payment): gold moving from a sender who is out, or to a receiver
# who has left, is a fallen ally's: inherited, never pool.
function FFP_Transfer
anchor set FF_st_poolOut[senderId] = FF_st_poolOut[senderId] + amount
replace if udg_player_active[senderId] and GetPlayerSlotState(target) == PLAYER_SLOT_STATE_PLAYING then
insert     set FF_st_poolOut[senderId] = FF_st_poolOut[senderId] + amount
anchor set FF_st_poolIn[targetId] = FF_st_poolIn[targetId] + amount
insert else
insert     set FF_st_inhOut[senderId] = FF_st_inhOut[senderId] + amount
insert     set FF_st_inhIn[targetId] = FF_st_inhIn[targetId] + amount
insert endif

# BASE collect gold (A08D on a leaver's base): the leaver's gold split over the active allies (ForForce callback).
function Trig_BASE_collect_gold_Func012Func003A
anchor call AdjustPlayerStateBJ(( udg_leaving_gold[udg_temp_playernumber] / udg_activeplayersinteam ), GetEnumPlayer(), PLAYER_STATE_RESOURCE_GOLD)
insert set FF_st_inhOut[udg_temp_playernumber] = FF_st_inhOut[udg_temp_playernumber] + udg_leaving_gold[udg_temp_playernumber] / udg_activeplayersinteam
insert set FF_st_inhIn[GetConvertedPlayerId(GetEnumPlayer())] = FF_st_inhIn[GetConvertedPlayerId(GetEnumPlayer())] + udg_leaving_gold[udg_temp_playernumber] / udg_activeplayersinteam

# BASE destroy (A06W) and player destroyed (base died, team alive): half to each living trio ally, or all to the
# one left. Same four lines in both functions.
function Trig_BASE_destroy_Actions
anchor call AdjustPlayerStateBJ(( udg_leaving_gold[udg_temp_playernumber] / 2 ), ConvertedPlayer(udg_temp_playernumber_ally_1), PLAYER_STATE_RESOURCE_GOLD)

insert set FF_st_inhIn[udg_temp_playernumber_ally_1] = FF_st_inhIn[udg_temp_playernumber_ally_1] + udg_leaving_gold[udg_temp_playernumber] / 2
anchor call AdjustPlayerStateBJ(( udg_leaving_gold[udg_temp_playernumber] / 2 ), ConvertedPlayer(udg_temp_playernumber_ally_2), PLAYER_STATE_RESOURCE_GOLD)
insert set FF_st_inhOut[udg_temp_playernumber] = FF_st_inhOut[udg_temp_playernumber] + udg_leaving_gold[udg_temp_playernumber] / 2
insert set FF_st_inhIn[udg_temp_playernumber_ally_2] = FF_st_inhIn[udg_temp_playernumber_ally_2] + udg_leaving_gold[udg_temp_playernumber] / 2
anchor call AdjustPlayerStateBJ(udg_leaving_gold[udg_temp_playernumber], ConvertedPlayer(udg_temp_playernumber_ally_1), PLAYER_STATE_RESOURCE_GOLD)
insert set FF_st_inhOut[udg_temp_playernumber] = FF_st_inhOut[udg_temp_playernumber] + udg_leaving_gold[udg_temp_playernumber]
insert set FF_st_inhIn[udg_temp_playernumber_ally_1] = FF_st_inhIn[udg_temp_playernumber_ally_1] + udg_leaving_gold[udg_temp_playernumber]
anchor call AdjustPlayerStateBJ(udg_leaving_gold[udg_temp_playernumber], ConvertedPlayer(udg_temp_playernumber_ally_2), PLAYER_STATE_RESOURCE_GOLD)
insert set FF_st_inhOut[udg_temp_playernumber] = FF_st_inhOut[udg_temp_playernumber] + udg_leaving_gold[udg_temp_playernumber]
insert set FF_st_inhIn[udg_temp_playernumber_ally_2] = FF_st_inhIn[udg_temp_playernumber_ally_2] + udg_leaving_gold[udg_temp_playernumber]

# King of the Hill income (every 3s, ForForce over the hill team): each player's actual payout.

# Spell dummies live until their spell's damage is over. A dummy that is gone when a later tick kills leaves no
# killer at all, so the game gives no kill, XP or gold. Each life = the spell's duration at the cast level + 1 s.
# Objection on an enemy: the dummy's Soul Burn.
function Trig_Objection_Actions
after call IssueTargetOrderBJ(udg_dummy, "soulburn", udg_temp_target)
anchor call UnitApplyTimedLifeBJ(0.10, 'BTLF', udg_dummy)
replace call UnitApplyTimedLifeBJ(RMaxBJ(BlzGetAbilityRealLevelField(BlzGetUnitAbility(udg_dummy, 'ANso'), ABILITY_RLF_DURATION_NORMAL, udg_abilitylevel - 1), BlzGetAbilityRealLevelField(BlzGetUnitAbility(udg_dummy, 'ANso'), ABILITY_RLF_DURATION_HERO, udg_abilitylevel - 1)) + 1.00, 'BTLF', udg_dummy)

# Jaina's ultimate: the three orbs vanish at the same moment as before, but stay alive (hidden, no longer casting)
